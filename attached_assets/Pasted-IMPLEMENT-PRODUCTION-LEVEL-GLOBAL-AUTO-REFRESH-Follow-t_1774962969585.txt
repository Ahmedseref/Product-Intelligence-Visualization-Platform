🔹 STEP 1 — CREATE GLOBAL REFRESH STATE (BACKEND)

Create a new file:

/backend/refresh_state.py

Add:

class RefreshState:
    def __init__(self):
        self.trigger_id = 0
        self.last_updated = None

    def trigger(self):
        from datetime import datetime
        self.trigger_id += 1
        self.last_updated = datetime.utcnow()

refresh_state = RefreshState()

🔹 STEP 4 — CREATE GLOBAL REFRESH HOOK (FRONTEND)

Create:

/frontend/hooks/useGlobalRefresh.js

Add:

import { useEffect, useState } from "react";

export default function useGlobalRefresh(onRefresh) {
  const [triggerId, setTriggerId] = useState(null);
  const [isEditing, setIsEditing] = useState(false);
  const [isFetching, setIsFetching] = useState(false);

  useEffect(() => {
    const interval = setInterval(async () => {
      if (isEditing || isFetching) return;

      setIsFetching(true);

      try {
        const res = await fetch("/api/refresh-state");
        const data = await res.json();

        if (triggerId !== null && data.trigger_id !== triggerId) {
          onRefresh();
        }

        setTriggerId(data.trigger_id);
      } catch (err) {
        console.error("Refresh error:", err);
      }

      setIsFetching(false);
    }, 30000); // every 30 sec

    return () => clearInterval(interval);
  }, [triggerId, isEditing]);

  return { setIsEditing };
}
